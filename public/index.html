// Makhazin landing page Worker // - Static files are served from ./public via the ASSETS binding // - POST /api/waitlist saves signups into the D1 database (binding: DB) const JSON_HEADERS = { "Content-Type": "application/json; charset=utf-8" }; function json(body, status = 200) { return new Response(JSON.stringify(body), { status, headers: JSON_HEADERS }); } const clean = (v, max) => String(v ?? "").trim().slice(0, max); export default { async fetch(request, env) { const url = new URL(request.url); if (url.pathname === "/api/waitlist") { if (request.method !== "POST") { return json({ error: "Method not allowed" }, 405); } let data; try { data = await request.json(); } catch { return json({ error: "Invalid request." }, 400); } // Honeypot: bots fill the hidden field, humans don't. Pretend success. if (clean(data.website, 200)) return json({ ok: true }); const role = data.role === "owner" ? "owner" : "tenant"; const name = clean(data.name, 120); const email = clean(data.email, 200).toLowerCase(); const phone = clean(data.phone, 30).replace(/[^\d+]/g, ""); const city = clean(data.city, 60); const sizeNum = parseInt(data.size, 10); const size = Number.isFinite(sizeNum) && sizeNum > 0 && sizeNum < 10000000 ? sizeNum : null; if (!name) return json({ error: "Please enter your name." }, 400); if (!email && !phone) { return json({ error: "Please add an email or a WhatsApp number." }, 400); } if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { return json({ error: "That email doesn't look right." }, 400); } if (phone && phone.replace(/\D/g, "").length < 8) { return json({ error: "That phone number looks too short." }, 400); } try { await env.DB.prepare( `INSERT INTO waitlist (role, name, email, phone, city, size_m2) VALUES (?1, ?2, ?3, ?4, ?5, ?6)` ) .bind(role, name, email || null, phone || null, city || null, size) .run(); } catch (err) { // Unique email/phone: treat a repeat signup as success if (String(err).includes("UNIQUE")) return json({ ok: true }); console.error("waitlist insert failed", err); return json({ error: "Something went wrong. Please try again." }, 500); } return json({ ok: true }); } return env.ASSETS.fetch(request); }, };
